## Create and Open a Collection

To create a new collection in Zvec, you need to define the following:

1. **Schema** —  the structural blueprint of your data, specifying scalar fields and vector embeddings.
1. **Collection options** (optional) — runtime settings that control how the collection behaves when opened (e.g., read-only mode).

Once defined, call `create_and_open()` to initialize a new collection at a specified path and return a `Collection` object ready for inserts and queries.

<Callout className="text-base" type="warn">
  If a collection already exists at the specified path, `create_and_open()` will raise an error to prevent accidental overwrites.
</Callout>

```python tab="Python" title="Create and open a collection" tab-group="code-demo"
import zvec

# [!code word:CollectionSchema]
# [!code word:CollectionOption]
# Define a collection schema
collection_schema = zvec.CollectionSchema(
    name="example_collection",
    fields=[
        zvec.FieldSchema(
            name="string_field_example",
            data_type=zvec.DataType.STRING,
            nullable=True,
            index_param=zvec.InvertIndexParam(enable_range_optimization=False),
        ),
    ],
    vectors=[
        zvec.VectorSchema(
            name="dense_vector_example",
            data_type=zvec.DataType.VECTOR_FP32,
            dimension=768,
            index_param=zvec.HnswIndexParam(metric_type=zvec.MetricType.COSINE),
        ),
    ],
)

# Create and open the collection
collection = zvec.create_and_open(  # [!code highlight]
    path="/path/to/my/collection",
    schema=collection_schema,
    option=zvec.CollectionOption(read_only=False, enable_mmap=True),
)
```

```ts tab="Node.js" title="Create and open a collection"
import { ZVecCollectionSchema, ZVecCreateAndOpen, ZVecDataType, ZVecIndexType, ZVecMetricType } from "@zvec/zvec";

// [!code word:ZVecCollectionSchema]
// Define a collection schema
const collectionSchema = new ZVecCollectionSchema({
    name: "example_collection",
    fields: [
        {
            name: "string_field_example",
            dataType: ZVecDataType.STRING,
            nullable: true,
            indexParams: { indexType: ZVecIndexType.INVERT, enableRangeOptimization: false }
        }
    ],
    vectors: [
        {
            name: "dense_vector_example",
            dataType: ZVecDataType.VECTOR_FP32,
            dimension: 768,
            indexParams: { indexType: ZVecIndexType.HNSW, metricType: ZVecMetricType.COSINE }
        }
    ]
});

// Create and open the collection
const collection = ZVecCreateAndOpen(       // [!code highlight]
    "/path/to/my/collection",
    collectionSchema,
    { readOnly: false, enableMMAP: true }   // [!code highlight]
);
```

```java tab="Java" title="Create and open a collection"
import org.zvec.binding.*;

// [!code word:CollectionSchema]
// [!code word:CollectionOptions]
// Define a collection schema
CollectionSchema collectionSchema = new CollectionSchema("example_collection");

try (FieldSchema stringField = new FieldSchema("string_field_example", DataType.STRING, true, 0);
     IndexParams invert = IndexParams.createInvert(false, false)) {
    stringField.setIndexParams(invert);
    collectionSchema.addField(stringField);
}

try (FieldSchema vectorField = new FieldSchema("dense_vector_example", DataType.VECTOR_FP32, false, 768);
     IndexParams hnsw = IndexParams.createHNSW(MetricType.COSINE, 16, 200)) {
    vectorField.setIndexParams(hnsw);
    collectionSchema.addField(vectorField);
}

// Create and open the collection
CollectionOptions options = new CollectionOptions();   // [!code highlight]
options.setReadOnly(false);   // [!code highlight]
options.setEnableMmap(true);   // [!code highlight]

Collection collection = Zvec.createAndOpen(   // [!code highlight]
        "/path/to/my/collection", collectionSchema, options);
collectionSchema.close();
```

## Real-World Example: 🛒 Product Search

This schema models a **multi-modal product search system**, combining visual, textual, and structured metadata for rich retrieval:

### 🗂️ Scalar Fields: For Filtering & Display

- `category` (array of strings, indexed): Enables queries like `category CONTAIN_ANY ("electronics", "headphones")` to find products that belong to either "electronics" or "headphones" (or both).
- `price` (integer, indexed with range optimization): Supports fast range queries such as `price > 100`.
- `in_stock` (boolean, indexed): Enables instant filtering by availability (e.g., "only show items in stock").
- `image_url` and `description` are stored but **not indexed**, since they're only used for display.

### 📐 Vector Embeddings: For Semantic Relevance

- Two dense vectors capture semantic meaning:
  - `image_vec`: 512-dimensional embeddings from product images (e.g., via a vision model).
  - `description_vec`: 768-dimensional embeddings from product descriptions (e.g., from a language model), stored with quantization.
- One sparse vector `keywords_sparse` for keyword matching, enabling hybrid sparse-dense search.

```python tab="Python" title="Create a collection" tab-group="code-demo"
import zvec

collection_schema = zvec.CollectionSchema(  # [!code highlight]
    name="product_search",
    fields=[  # [!code highlight]
        zvec.FieldSchema(
            name="image_url",
            data_type=zvec.DataType.STRING,  # Not used in filtering, no index created
            nullable=True,  # Could be null
        ),
        zvec.FieldSchema(
            name="description",
            data_type=zvec.DataType.STRING,  # Not used in filtering, no index created
        ),
        zvec.FieldSchema(
            name="category",
            data_type=zvec.DataType.ARRAY_STRING,
            # Inverted index for array membership queries
            index_param=zvec.InvertIndexParam(),
        ),
        zvec.FieldSchema(
            name="price",
            data_type=zvec.DataType.INT32,
            # Optimization for range queries, e.g., price > 100
            index_param=zvec.InvertIndexParam(enable_range_optimization=True),
        ),
        zvec.FieldSchema(
            name="in_stock",
            data_type=zvec.DataType.BOOL,
            # Inverted index for boolean queries
            index_param=zvec.InvertIndexParam(),
        ),
    ],
    vectors=[  # [!code highlight]
        # Dense embedding from product images
        zvec.VectorSchema(
            name="image_vec",
            data_type=zvec.DataType.VECTOR_FP32,
            dimension=512,
            # Use HNSW index for similarity search with cosine distance metric
            index_param=zvec.HnswIndexParam(metric_type=zvec.MetricType.COSINE),
        ),
        # Dense embedding from product descriptions
        zvec.VectorSchema(
            name="description_vec",
            data_type=zvec.DataType.VECTOR_FP32,
            dimension=768,
            # Enable quantization for faster similarity search
            index_param=zvec.HnswIndexParam(metric_type=zvec.MetricType.COSINE, quantize_type=zvec.QuantizeType.INT8),
        ),
        # Sparse vector from product keywords
        zvec.VectorSchema(
            name="keywords_sparse",
            data_type=zvec.DataType.SPARSE_VECTOR_FP32,
            # Use HNSW index for similarity search with inner product metric
            index_param=zvec.HnswIndexParam(metric_type=zvec.MetricType.IP),
        ),
    ],
)

collection = zvec.create_and_open(  # [!code highlight]
    path="path/to/collection",
    schema=collection_schema,
    option=zvec.CollectionOption(read_only=False, enable_mmap=True),
)
```

```java tab="Java" title="Create a collection"
import org.zvec.binding.*;

CollectionSchema collectionSchema = new CollectionSchema("product_search");   // [!code highlight]

// Scalar fields
try (FieldSchema imageUrl = new FieldSchema("image_url", DataType.STRING, true, 0)) {
    collectionSchema.addField(imageUrl);   // Not used in filtering, no index created
}

try (FieldSchema description = new FieldSchema("description", DataType.STRING, true, 0)) {
    collectionSchema.addField(description);   // Not used in filtering, no index created
}

try (FieldSchema category = new FieldSchema("category", DataType.ARRAY_STRING, true, 0);
     IndexParams invert = IndexParams.createInvert(false, false)) {
    // Inverted index for array membership queries
    category.setIndexParams(invert);
    collectionSchema.addField(category);
}

try (FieldSchema price = new FieldSchema("price", DataType.INT32, true, 0);
     IndexParams invert = IndexParams.createInvert(true, false)) {
    // Optimization for range queries, e.g., price > 100
    price.setIndexParams(invert);
    collectionSchema.addField(price);
}

try (FieldSchema inStock = new FieldSchema("in_stock", DataType.BOOL, true, 0);
     IndexParams invert = IndexParams.createInvert(false, false)) {
    // Inverted index for boolean queries
    inStock.setIndexParams(invert);
    collectionSchema.addField(inStock);
}

// Vector fields
// Dense embedding from product images
try (FieldSchema imageVec = new FieldSchema("image_vec", DataType.VECTOR_FP32, false, 512);
     IndexParams hnsw = IndexParams.createHNSW(MetricType.COSINE, 16, 200)) {
    // Use HNSW index for similarity search with cosine distance metric
    imageVec.setIndexParams(hnsw);
    collectionSchema.addField(imageVec);
}

// Dense embedding from product descriptions
try (FieldSchema descriptionVec = new FieldSchema("description_vec", DataType.VECTOR_FP32, false, 768);
     IndexParams hnsw = IndexParams.createHNSWQuantized(MetricType.COSINE, 16, 200, QuantizeType.INT8)) {
    // Enable quantization for faster similarity search
    descriptionVec.setIndexParams(hnsw);
    collectionSchema.addField(descriptionVec);
}

// Sparse vector from product keywords
try (FieldSchema keywordsSparse = new FieldSchema("keywords_sparse", DataType.SPARSE_VECTOR_FP32, false, 0);
     IndexParams hnsw = IndexParams.createHNSW(MetricType.IP, 16, 200)) {
    // Use HNSW index for similarity search with inner product metric
    keywordsSparse.setIndexParams(hnsw);
    collectionSchema.addField(keywordsSparse);
}

CollectionOptions options = new CollectionOptions();
options.setReadOnly(false);
options.setEnableMmap(true);

Collection collection = Zvec.createAndOpen("path/to/collection", collectionSchema, options);   // [!code highlight]
collectionSchema.close();
```
