The `query()` method supports **vector similarity search**, **full-text search** (BM25 ranking), **conditional filtering** (like a SQL `WHERE` clause), or **combinations of these**.

It returns a list of `Doc` objects, each containing the matched [document](../../concepts/data-modeling/#documents) and its relevance score.

---

## `Query`

In Zvec, a `Query` is a query specification that describes the field and query source used for a search. Query APIs accept one or more `Query` specifications depending on whether you are performing a single- or multi-query search.

<Callout className="text-base" type="info">
  A single `Query` can target either vector search or full-text search, but not both at the same time.
</Callout>

<Tabs items={['Python', 'Node.js', 'Java']}>

<Tab value="Python">

Each `Query` specifies:

1. `field_name`: The name of the vector or full-text field to search
1. **Query source**:
      - For vector search, provide an explicit `vector` or a document `id` (to reuse the stored embedding of an existing document)
      - For full-text search, provide an `fts` clause
1. `param` (optional): Index-specific query parameters (e.g., `ef` for [HNSW](../../concepts/vector-index/hnsw-index/#query-time-parameters) or `default_operator` for [full-text search](./fts/#default-operator))

<Accordions type="single">

<Accordion title="Code Example">

```python title="Query"
import zvec
from zvec.model.param.query import Fts, Query

vector_query = Query(  # [!code highlight]
    field_name="dense_embedding",
    vector=[0.1] * 768,  # Use real embedding in practice
)

by_id_query = Query(  # [!code highlight]
    field_name="dense_embedding",
    id="doc123",  # Use the 'dense_embedding' from the document with ID "doc123"
)

fts_query = Query(  # [!code highlight]
    field_name="content",
    fts=Fts(match_string="machine learning"),
)
```

1. **Query source**:

1. `params` (optional): Index-specific query parameters (e.g., `ef` for [HNSW](../../concepts/vector-index/hnsw-index/#query-time-parameters) or `defaultOperator` for [full-text search](./fts/#default-operator))

<Tab value="Java">

Each `VectorQuery` specifies:

1. `fieldName`: The name of the vector or full-text field to search
1. **Query source**:
      - For vector search, provide an explicit query `vector`
      - For full-text search, provide an `FtsPayload`
1. Query parameters (optional): Index-specific query parameters (e.g., the `ef` of `HnswQueryParams` for [HNSW](../../concepts/vector-index/hnsw-index/#query-time-parameters), or the `defaultOperator` of `FtsQueryParams` for [full-text search](./fts/#default-operator))

<Accordions type="single">

<Accordion title="Code Example">

```java title="VectorQuery"
import java.util.Arrays;
import org.zvec.binding.*;

float[] vector = new float[768];
Arrays.fill(vector, 0.1f);  // Use real embedding in practice

VectorQuery vectorQuery = new VectorQuery();   // [!code highlight]
vectorQuery.setFieldName("dense_embedding");
vectorQuery.setQueryVector(vector);

// [!code word:fts]
VectorQuery ftsQuery = new VectorQuery();   // [!code highlight]
ftsQuery.setFieldName("content");
try (FtsPayload fts = new FtsPayload()) {
    fts.setMatchString("machine learning");
    ftsQuery.setFts(fts);
}
```

## Quick Start Examples

### Single-Vector Search

```python tab="Python" tab-group="code-demo"
import zvec

result = collection.query(  # [!code highlight]
    queries=zvec.Query(
        field_name="dense_embedding",
        vector=[0.1] * 768,  # Use real embedding in practice
    ),
    topk=10,
)
```

```ts tab="Node.js"
let result = collection.querySync({   // [!code highlight]
    fieldName: "dense_embedding",
    vector: Array(768).fill(0.1),   // Use real embedding in practice
    topk: 10,
});
```

```java tab="Java"
import java.util.Arrays;
import java.util.List;
import org.zvec.binding.*;

float[] vector = new float[768];
Arrays.fill(vector, 0.1f);   // Use real embedding in practice

try (VectorQuery query = new VectorQuery()) {   // [!code highlight]
    query.setFieldName("dense_embedding");
    query.setQueryVector(vector);
    query.setTopK(10);

    List<Doc> result = collection.query(query);
    Doc.freeDocs(result);
}
```

### Multi-Vector Search

```python tab="Python" tab-group="code-demo"
import zvec

result = collection.query(  # [!code highlight]
    topk=3,
    queries=[
        zvec.Query(field_name="dense_embedding", vector=[0.1] * 768),
        zvec.Query(field_name="sparse_embedding", vector={1: 0.1, 37: 0.43}),
    ],
    reranker=zvec.WeightedReRanker(  # [!code highlight]
        weights=[1.2, 1.0],
    ),
)
print(result)
```

```ts tab="Node.js" tab-group="code-demo"
let result = collection.multiQuerySync({  // [!code highlight]
    topk: 3,
    queries: [
        { fieldName: "dense_embedding", vector: Array(768).fill(0.1), numCandidates: 5 },
        { fieldName: "sparse_embedding", vector: { 1: 0.1, 37: 0.43 }, numCandidates: 5 },
    ],
    rerank: { type: "weighted", weights: [1.2, 1.0] },  // [!code highlight]
});
```

```java tab="Java"
import java.util.Arrays;
import java.util.List;
import org.zvec.binding.*;

float[] denseVector = new float[768];
Arrays.fill(denseVector, 0.1f);

try (MultiQuery query = new MultiQuery();   // [!code highlight]
     SubQuery dense = new SubQuery();
     SubQuery sparse = new SubQuery()) {
    dense.setFieldName("dense_embedding");
    dense.setQueryVector(denseVector);
    dense.setNumCandidates(5);
    query.addSubQuery(dense);

    sparse.setFieldName("sparse_embedding");
    sparse.setSparseVector(new int[]{1, 37}, new float[]{0.1f, 0.43f});
    sparse.setNumCandidates(5);
    query.addSubQuery(sparse);

    query.setTopk(3);
    query.setRerankWeighted(new double[]{1.2, 1.0});   // [!code highlight]

    List<Doc> result = collection.query(query);
    Doc.freeDocs(result);
}
```

### Conditional Filtering

```python tab="Python" tab-group="code-demo"

result = collection.query(filter="publish_year < 1999", topk=50)
```

```ts tab="Node.js"
// [!code word:filter]

```

```java tab="Java"
// [!code word:setFilter]

    query.setFilter("publish_year < 1999");
    query.setTopK(50);

    List<Doc> result = collection.query(query);
    Doc.freeDocs(result);
}
```

### Grouped Search

```python tab="Python"
import zvec

groups = collection.group_by_query(  # [!code highlight]
    query=zvec.Query(
        field_name="dense_embedding",
        vector=[0.1] * 768,  # Replace with a real embedding in practice
    ),
    group_by_field_name="publish_year",  # Group by publication year
    group_count=3,                        # Return up to 3 groups
    topk_per_group=2,                     # Return up to 2 documents per group
)

for group in groups:
    print(group.group_by_value, group.docs)
```

```ts tab="Node.js"
const groups = collection.groupByQuerySync({  // [!code highlight]
    fieldName: "dense_embedding",
    vector: Array(768).fill(0.1),  // Replace with a real embedding in practice
    groupByFieldName: "publish_year",  // Group by publication year
    groupCount: 3,                      // Return up to 3 groups
    topkPerGroup: 2,                    // Return up to 2 documents per group
});

for (const group of groups) {
    console.log(group.groupByValue, group.docs);
}
```
