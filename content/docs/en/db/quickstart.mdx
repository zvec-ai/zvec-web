## Create a Collection

A [collection](../collections/) stores your documents. Each [document](../concepts/data-modeling/#documents) contains scalar fields and [vector embeddings](../concepts/vector-embedding/).

Define a schema and create a collection. A schema has two parts: `fields` for scalar data and `vectors` for vector embeddings.

```python tab="Python" title="Create a collection" tab-group="code-demo"
import zvec

# [!code word:embedding]
# [!code word:publish_year]
# Define a collection schema
collection_schema = zvec.CollectionSchema(  # [!code highlight]
    name="my_collection",
    fields=[
        zvec.FieldSchema(
            name="publish_year",
            data_type=zvec.DataType.INT32,
            index_param=zvec.InvertIndexParam(enable_range_optimization=True),
        ),
    ],
    vectors=[
        zvec.VectorSchema(
            name="embedding",
            data_type=zvec.DataType.VECTOR_FP32,
            dimension=768,
            index_param=zvec.HnswIndexParam(metric_type=zvec.MetricType.COSINE),
        ),
    ],
)

# Create a collection
collection = zvec.create_and_open(  # [!code highlight]
    path="./my_collection_data",
    schema=collection_schema,
)
```

```ts tab="Node.js" title="Create a collection"
import { ZVecCollectionSchema, ZVecCreateAndOpen, ZVecDataType, ZVecIndexType, ZVecMetricType } from "@zvec/zvec"

// [!code word:embedding]
// [!code word:publish_year]
// Define a collection schema
const collectionSchema = new ZVecCollectionSchema({     // [!code highlight]
    name: "my_collection",
    fields: [
        {
            name: "publish_year",
            dataType: ZVecDataType.INT32,
            indexParams: { indexType: ZVecIndexType.INVERT, enableRangeOptimization: true }
        }
    ],
    vectors: [
        {
            name: "embedding",
            dataType: ZVecDataType.VECTOR_FP32,
            dimension: 768,
            indexParams: { indexType: ZVecIndexType.HNSW, metricType: ZVecMetricType.COSINE }
        }
    ]
});

// Create a collection
const collection = ZVecCreateAndOpen("./my_collection_data", collectionSchema);   // [!code highlight]
```

```java tab="Java" title="Create a collection"
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import org.zvec.binding.*;

// [!code word:embedding]
// [!code word:publish_year]
// Define a collection schema
CollectionSchema collectionSchema = new CollectionSchema("my_collection");   // [!code highlight]

try (FieldSchema publishYear = new FieldSchema("publish_year", DataType.INT32, true, 0);
     IndexParams invert = IndexParams.createInvert(true, false)) {
    publishYear.setIndexParams(invert);
    collectionSchema.addField(publishYear);
}

try (FieldSchema embedding = new FieldSchema("embedding", DataType.VECTOR_FP32, false, 768);
     IndexParams hnsw = IndexParams.createHNSW(MetricType.COSINE, 16, 200)) {
    embedding.setIndexParams(hnsw);
    collectionSchema.addField(embedding);
}

// Create a collection
Zvec.initialize(null);
Collection collection = Zvec.createAndOpen("./my_collection_data", collectionSchema, null);   // [!code highlight]
collectionSchema.close();
```

<Callout className="text-base" type="info">
  **Important**: The field names you define here (`publish_year`, `embedding`) must be used exactly as written when inserting or querying data.
</Callout>

## Add Documents

[Insert](../data-operations/insert/) documents with scalar fields and vector embeddings:

```python tab="Python" title="Insert a document" tab-group="code-demo"
# [!code word:embedding]
# [!code word:publish_year]
collection.insert(  # [!code highlight]
    zvec.Doc(
        id="book_1",  # Unique document ID
        vectors={"embedding": [0.1] * 768}, # Replace with your actual vector
        fields={"publish_year": 1936},
    )
)
```

```ts tab="Node.js" title="Insert a document"
// [!code word:embedding]
// [!code word:publish_year]
collection.insertSync({     // [!code highlight]
    id: "book_1",   // Unique document ID
    vectors: { "embedding": Array(768).fill(0.1) },   // Replace with your actual vector
    fields: { "publish_year": 1936 }
});
```

```java tab="Java" title="Insert a document"
// [!code word:embedding]
// [!code word:publish_year]
float[] embedding = new float[768];
Arrays.fill(embedding, 0.1f);   // Replace with your actual vector

Doc doc = new Doc();
doc.setPK("book_1");   // Unique document ID
doc.addVectorFP32Field("embedding", embedding);
doc.addInt32Field("publish_year", 1936);

List<Doc> docs = Collections.singletonList(doc);
Collection.WriteResult result = collection.insert(docs);   // [!code highlight]
System.out.println(result.successCount);
Doc.freeDocs(docs);
```

<Callout className="text-base" type="info">
  **Important**: Field names must match exactly. The `publish_year` field and `embedding` vector must use the same names you defined in your schema.
</Callout>

## Optimize a Collection

New vectors are staged in a temporary index. Call [`optimize()`](../collections/optimize/) to build the vector index for faster search:

```python tab="Python" title="Optimize a collection" tab-group="code-demo"
# [!code word:optimize]
collection.optimize()
```

```ts tab="Node.js" title="Optimize a collection"
// [!code word:optimizeSync]
// Sync
collection.optimizeSync();

// [!code word:optimize]
// Async
await collection.optimize();
```

```java tab="Java" title="Optimize a collection"
// [!code word:optimize]
collection.optimize();
```

## Retrieve a Document by ID

[Fetch](../data-operations/fetch/) a document directly by its `id`:

```python tab="Python" title="Fetch a document" tab-group="code-demo"
result = collection.fetch(ids="book_1")   # [!code highlight]
print(result)
```

```ts tab="Node.js" title="Fetch a document"
let result = collection.fetchSync("book_1");  // [!code highlight]

```

```java tab="Java" title="Fetch a document"
List<Doc> result = collection.fetch(Collections.singletonList("book_1"));   // [!code highlight]
System.out.println(result.get(0).toDetailString());
```

## Search with Vectors

### Basic Similarity Search

Use [`query()`](../data-operations/query/) to find documents most similar to a given vector embedding:

```python tab="Python" title="Basic similarity search" tab-group="code-demo"
# [!code word:embedding]
result = collection.query(  # [!code highlight]
    queries=zvec.Query(
        field_name="embedding",
        vector=[0.3] * 768,  # Replace with your actual vector
    ),
    topk=10,
)
print(result)
```

```ts tab="Node.js" title="Basic similarity search"
// [!code word:embedding]
// Sync
let result = collection.querySync({   // [!code highlight]
    fieldName: "embedding",
    vector: Array(768).fill(0.3),     // Replace with your actual vector
    topk: 10
});
console.log(result);

// Async
let resultAsync = await collection.query({   // [!code highlight]
    fieldName: "embedding",
    vector: Array(768).fill(0.3),     // Replace with your actual vector
    topk: 10
});
console.log(resultAsync);
```

```java tab="Java" title="Basic similarity search"
// [!code word:embedding]
float[] vector = new float[768];
Arrays.fill(vector, 0.3f);   // Replace with your actual vector

try (VectorQuery query = new VectorQuery()) {
    query.setFieldName("embedding");
    query.setQueryVector(vector);
    query.setTopK(10);

    List<Doc> result = collection.query(query);   // [!code highlight]
    for (Doc hit : result) {
        System.out.printf("id=%s, score=%.5f%n", hit.getPK(), hit.getScore());
    }
    Doc.freeDocs(result);
}
```

### Filtered Similarity Search

Combine vector search with conditional filters — only matching documents are considered during search:

```python tab="Python" title="Filtered similarity search" tab-group="code-demo"
# [!code word:embedding]
result = collection.query(        # [!code highlight]
    queries=zvec.Query(
        field_name="embedding",
        vector=[0.3] * 768,   # Replace with your actual vector
    ),
    topk=10,
    filter="publish_year > 1936", # [!code highlight]
)
print(result)
```

```ts tab="Node.js" title="Filtered similarity search"
// [!code word:embedding]
// Sync
let result = collection.querySync({     // [!code highlight]
    fieldName: "embedding",
    vector: Array(768).fill(0.3),   // Replace with your actual vector
    topk: 10,
    filter: "publish_year > 1936"       // [!code highlight]
});
console.log(result);

// Async
let resultAsync = await collection.query({     // [!code highlight]
    fieldName: "embedding",
    vector: Array(768).fill(0.3),   // Replace with your actual vector
    topk: 10,
    filter: "publish_year > 1936"       // [!code highlight]
});
console.log(resultAsync);
```

```java tab="Java" title="Filtered similarity search"
// [!code word:embedding]
float[] vector = new float[768];
Arrays.fill(vector, 0.3f);   // Replace with your actual vector

try (VectorQuery query = new VectorQuery()) {
    query.setFieldName("embedding");
    query.setQueryVector(vector);
    query.setTopK(10);
    query.setFilter("publish_year > 1936");   // [!code highlight]

    List<Doc> result = collection.query(query);   // [!code highlight]
    for (Doc hit : result) {
        System.out.printf("id=%s, score=%.5f%n", hit.getPK(), hit.getScore());
    }
    Doc.freeDocs(result);
}
```

## Inspect a Collection

View the collection's schema:

```python tab="Python" title="View collection schema" tab-group="code-demo"
# [!code word:schema]
print(collection.schema)
```

```ts tab="Node.js" title="View collection schema"
// [!code word:schema]
console.log(collection.schema.toString());
```

```java tab="Java" title="View collection schema"
// [!code word:schema]
try (CollectionSchema schema = collection.getSchema()) {   // [!code highlight]
    System.out.println(schema.getName() + " " + schema.getAllFieldNames());
}
```

View the collection's statistics:

```python tab="Python" title="View collection statistics" tab-group="code-demo"
# [!code word:stats]

```

```ts tab="Node.js" title="View collection statistics"
// [!code word:stats]

```

```java tab="Java" title="View collection statistics"
// [!code word:stats]
CollectionStats stats = collection.getStats();
System.out.println(stats);
```
